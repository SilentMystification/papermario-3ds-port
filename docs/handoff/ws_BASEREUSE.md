def base_of(c, b):  # BASEREUSE: the stored commit whose base of b holds for c: c, else the newest stored commit with the same inputs
    if not stored_why(c, b): return c
    idx = load_index()["commits"]
    return next((s for s in reversed(list(idx)) if s != c and b in idx[s]["benches"] and not stored_why(s, b) and inputs_equal(s, c)), None)
def base_alias(c, b):  # BASEREUSE: True when a copy gave c its base of b
    s = base_of(c, b)
    if not s or s == c: return False
    copy_entry(s, c, b, {"alias_of": s}); print("base %s at %s: same inputs as %s: copied. No build, no run." % (b, c[:9], s[:9])); return True
def adopt_why(rec, tip, par, n):  # BASEREUSE: "" when the cmp pass rec may give its bench a base at the merge, else the reason
    b = rec["bench"]; run_c = base_commit(rec["run_commit"]) if rec.get("run_commit") else None; pb = base_of(par, b)
    if n >= BASE_ADOPT_MAX: return "full base due (%d adoptions since the last full base)" % n
    if rec.get("tools") != tools_hash(): return "the measure tools changed since the cmp"
    if rec.get("dirty") is not False: return "the row run had uncommitted changes (dirty %s)" % rec.get("dirty")
    if not (tip and run_c and inputs_equal(run_c, tip)): return "the row run is not at the inputs of the branch tip"
    if not pb: return "main before the merge (%s) has no stored base" % par[:9]
    if shot_sha(pb, b) != rec.get("shots"): return "the base at %s differs from the base of the row cmp" % pb[:9]
    if json.loads((store_root() / pb / b / "meta.json").read_text(encoding="utf-8")).get("tools") != rec["tools"]: return "the base at %s was made by other measure tools" % pb[:9]
    return ""
